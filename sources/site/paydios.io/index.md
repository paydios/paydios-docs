### React / Next.js

Clean, production-ready implementation. Copy the integration snippet directly into your project to start routing transactions.

InstallationCopy

pnpm add @paydios/react

Verified IntegrationPCI-DSS Level 1

CheckoutModal.tsx

Copy Snippet

```
import { PaydiosCheckout, usePaydios } from "@paydios/react";

export function PaymentCheckout({ orderId, amount }: Props) {
  const { initiatePayment, isLoading } = usePaydios();

  const handlePay = async () => {
    await initiatePayment({
      orderId,
      amount,
      currency: "USD",
      onSuccess: (tx) => console.log("Settled on-chain:", tx.id),
      onError: (err) => console.error("Payment failed:", err.message),
    });
  };

  return (
    <PaydiosCheckout
      theme="dark"
      amount={amount}
      onPay={handlePay}
      disabled={isLoading}
      label="Instant Checkout with Paydios"
    />
  );
}
```
